1. Input Sanitization & Tokenization

Demonstrates efficient string manipulation and preparation for algorithmic processing.

--!strict
type TokenArray = {string}

local function sanitizeAndTokenize(rawInput: string): TokenArray
    -- 1. Normalize to lowercase
    local cleanInput: string = string.lower(rawInput)

    -- 2. Strip punctuation and special characters using pattern matching
    -- %p matches punctuation, %c matches control characters
    cleanInput = string.gsub(cleanInput, "[%p%c]", "")

-- 3. Tokenize into an array of words
    local tokens: TokenArray = {}
    for word in string.gmatch(cleanInput, "%S+") do
        table.insert(tokens, word)
    end

    return tokens
end

2. The Heuristic Scoring Core

Demonstrates algorithmic logic, nested iteration, and performance optimization via early-exit constraints.

type ResponseTemplate = {
    TopicName: string,
    RequiredWords: {string},
    UserInputKeywords: {string},
    StandardResponses: {string},
    RepeatedResponses: {string}
}

local function evaluateBestResponse(tokens: TokenArray, database: {ResponseTemplate}): ResponseTemplate?
    local bestScore: number = 0
    local bestTemplate: ResponseTemplate? = nil

    for _, template in ipairs(database) do
        -- OPTIMIZATION: Early exit. If required words aren't present, skip scoring entirely.
        local hasAllRequired: boolean = true
        for _, reqWord in ipairs(template.RequiredWords) do
            if not table.find(tokens, reqWord) then
                hasAllRequired = false
                break
            end
        end

        if hasAllRequired then
            local currentScore: number = 0

            -- Score based on additional keyword density
            for _, token in ipairs(tokens) do
                if table.find(template.UserInputKeywords, token) then
                    currentScore += 1
                end
            end

            -- Update best match if this template scored higher
            if currentScore > bestScore then
                bestScore = currentScore
                bestTemplate = template
            end
        end
    end

    -- Returns nil if no template scored > 0, triggering a fallback response
    return bestTemplate
end

3. State Management & Anti-Spam Logic

Demonstrates memory management, conversational continuity, and defensive programming against abuse.

local SessionState = {
    CurrentTopic = "Idle",
    RepetitionCount = 0,
    Sentiment = { Anger = 0, Friendship = 0 }
}

local THRESHOLDS = {
    WARNING = 3,
    KICK = 5
}

local function processConversationState(selectedTemplate: ResponseTemplate, player: Player): string
    local newTopic = selectedTemplate.TopicName

    if SessionState.CurrentTopic == newTopic then
        SessionState.RepetitionCount += 1

        -- Defensive: Prevent infinite loops or database spam
        if SessionState.RepetitionCount >= THRESHOLDS.KICK then
            warn(`[Security] Spam detected from {player.Name}. Terminating session.`)
            player:Kick("Excessive conversational repetition detected.")
            return ""
        elseif SessionState.RepetitionCount >= THRESHOLDS.WARNING then
            -- Provide a specialized "stop spamming" response
            return selectedTemplate.RepeatedResponses[math.random(1, #selectedTemplate.RepeatedResponses)]
        end
    else
        -- Topic Shift: Reset counters and update context
        SessionState.CurrentTopic = newTopic
        SessionState.RepetitionCount = 1

        -- Example of sentiment tracking affecting state
        if newTopic == "Compliment" then
            SessionState.Sentiment.Friendship += 1
            SessionState.Sentiment.Anger = math.clamp(SessionState.Sentiment.Anger - 1, 0, 10)
        elseif newTopic == "Insult" then
            SessionState.Sentiment.Anger += 1
        end
    end

    -- Return a standard randomized response from the matched topic
    return selectedTemplate.StandardResponses[math.random(1, #selectedTemplate.StandardResponses)]
end

4. Dynamic Mode Switching (Safe Math Evaluation)

Demonstrates the ability to intercept standard flow for specialized tasks, and crucially, the use of pcall to prevent code injection or crashes.

local function handleDynamicModes(tokens: TokenArray, rawInput: string): string?
    -- Detect if input is purely a math equation (contains digits AND operators)
    local hasNumber: boolean = string.match(rawInput, "%d") ~= nil
    local hasOperator: boolean = string.match(rawInput, "[%+%-%*%/%%]") ~= nil

    if hasNumber and hasOperator and SessionState.CurrentTopic == "Idle" then
        SessionState.CurrentTopic = "MathMode" -- Lock state

        -- Sanitize strictly to only allow math characters before evaluation
        local safeEquation: string = string.gsub(rawInput, "[^%d%+%-%*%/%%%.%s]", "")

        -- SAFE EXECUTION: Wrap dynamic code evaluation in pcall (protected call)
        local success: boolean, result: any = pcall(function()
            local evalFunc = loadstring("return " .. safeEquation)
            return evalFunc and evalFunc() or "Invalid Expression"
        end)

        SessionState.CurrentTopic = "Idle" -- Unlock state

        if success and type(result) == "number" then
            return `The calculated result is: {result}`
        else
            return "I encountered an error computing that equation."
        end
    end

    return nil -- Fall back to standard heuristic scoring
end

5. The Main Orchestrator

Demonstrates how the components are cleanly composed into a single, readable event listener.

local ChatService = game:GetService("Chat")

game.Players.PlayerAdded:Connect(function(player: Player)
    player.Chatted:Connect(function(rawMessage: string)
        -- 1. Sanitize
        local tokens = sanitizeAndTokenize(rawMessage)

        -- 2. Check for Dynamic Modes first (e.g., Math, Commands)
        local modeResponse = handleDynamicModes(tokens, rawMessage)
        if modeResponse then
            ChatService:Chat(player.Character.Head, modeResponse, Enum.ChatColor.Blue)
            return
        end

        -- 3. Heuristic Scoring
        local bestMatch = evaluateBestResponse(tokens, ResponseDatabase)

        -- 4. Generate Reply (or fallback)
        local finalReply: string
        if bestMatch then
            finalReply = processConversationState(bestMatch, player)
        else
            finalReply = FallbackDatabase[math.random(1, #FallbackDatabase)]
            SessionState.CurrentTopic = "Idle" -- Reset on misunderstanding
        end

        -- 5. Output to player
        if finalReply ~= "" then
            ChatService:Chat(player.Character.Head, finalReply, Enum.ChatColor.Blue)
        end
    end)
end)
